## Symptom

Tool calls fail with 404 or connection errors. The agent confidently references endpoints like `GET /api/v2/users/search` when the actual path is `/api/v1/user/find`. After a failed call, the agent invents a slightly different URL and retries, entering a loop. The agent cites documentation URLs that return 404. API integration code generated by the agent doesn't work without manual correction.

## Root Cause

Claude's training data contains many API patterns, so it can generate plausible-sounding endpoint structures. When the actual API reference is not in context, Claude interpolates from patterns it has seen — producing syntactically valid but semantically wrong URLs. Version numbers (`/v1/` vs `/v2/`), plural vs singular resource names, and query parameter names are common hallucination points. The agent has no mechanism to verify a URL before using it.

## Fix

### Option 1: Inject API reference documentation into system prompt

```python
import anthropic

client = anthropic.Anthropic()

# Real API spec — injected at system prompt level, cached for efficiency
WEATHER_API_SPEC = """
Base URL: https://api.weatherservice.example.com

Endpoints:
  GET /v1/current?city={city}&units={metric|imperial}
    Returns: {"temp": float, "condition": str, "humidity": int}

  GET /v1/forecast?city={city}&days={1-7}&units={metric|imperial}
    Returns: {"forecast": [{"date": str, "high": float, "low": float}]}

  GET /v1/alerts?city={city}
    Returns: {"alerts": [{"severity": str, "message": str}]}

Authentication: Pass API key as header: X-API-Key: {key}
Rate limit: 100 requests/minute
"""

SYSTEM_PROMPT = f"""You are an assistant that can query weather data.

<api_reference>
{WEATHER_API_SPEC}
</api_reference>

<instructions>
- Only use endpoints listed in <api_reference>. Do not invent endpoints.
- Always include the required query parameters shown.
- If unsure about a parameter value, ask the user rather than guessing.
- Report the exact URL you would call before calling it.
</instructions>"""

TOOLS = [
    {
        "name": "call_weather_api",
        "description": "Call the weather API. Only use endpoints from the API reference in the system prompt.",
        "input_schema": {
            "type": "object",
            "properties": {
                "endpoint": {
                    "type": "string",
                    "description": "Exact endpoint path from the API reference (e.g., /v1/current)",
                },
                "params": {
                    "type": "object",
                    "description": "Query parameters as key-value pairs",
                },
            },
            "required": ["endpoint", "params"],
        },
    }
]


def call_weather_api(endpoint: str, params: dict) -> str:
    # Validate endpoint against known list before making real call
    known_endpoints = {"/v1/current", "/v1/forecast", "/v1/alerts"}
    if endpoint not in known_endpoints:
        return f"ERROR: Unknown endpoint '{endpoint}'. Valid endpoints: {', '.join(sorted(known_endpoints))}"

    # Simulate API call
    return f"Simulated response for {endpoint} with params {params}"


response = client.messages.create(
    model="claude-sonnet-4-6",
    max_tokens=512,
    system=SYSTEM_PROMPT,
    tools=TOOLS,
    messages=[{"role": "user", "content": "What's the 3-day forecast for Tokyo?"}],
)

for block in response.content:
    if block.type == "tool_use":
        print(f"Tool call: {block.name}")
        print(f"Endpoint: {block.input.get('endpoint')}")
        print(f"Params: {block.input.get('params')}")
        result = call_weather_api(block.input["endpoint"], block.input.get("params", {}))
        print(f"Result: {result}")
```

**Expected Token Savings:** Grounded endpoints eliminate retry loops from 404 errors — each failed call costs a full round-trip.
**Environment:** Python 3.9+; cache the API spec with `cache_control: {"type": "ephemeral"}` for multi-turn sessions.

### Option 2: URL validation tool that checks endpoints before calling them

```python
import re
import anthropic
from urllib.parse import urlparse

client = anthropic.Anthropic()

# Allowlist of valid endpoint patterns (compiled regexes for speed)
VALID_ENDPOINT_PATTERNS = [
    re.compile(r"^/v1/users(/\w+)?$"),
    re.compile(r"^/v1/users/\w+/profile$"),
    re.compile(r"^/v1/teams(/\w+)?$"),
    re.compile(r"^/v1/messages$"),
    re.compile(r"^/v1/messages/\w+$"),
    re.compile(r"^/v2/search$"),
    re.compile(r"^/v2/analytics/events$"),
]

BASE_URL = "https://api.example.com"


def validate_endpoint(path: str) -> tuple[bool, str]:
    """Check if an endpoint path matches any known pattern."""
    # Normalize path
    path = path.strip()
    if not path.startswith("/"):
        path = "/" + path

    for pattern in VALID_ENDPOINT_PATTERNS:
        if pattern.match(path):
            return True, f"Valid endpoint: {path}"

    return False, (
        f"Unknown endpoint: '{path}'. "
        f"This endpoint does not exist in the API. "
        f"Valid prefixes: /v1/users, /v1/teams, /v1/messages, /v2/search, /v2/analytics"
    )


TOOLS = [
    {
        "name": "validate_api_endpoint",
        "description": "Check if an API endpoint path is valid before calling it. Always use this before call_api.",
        "input_schema": {
            "type": "object",
            "properties": {
                "path": {"type": "string", "description": "API endpoint path to validate, e.g. /v1/users"},
            },
            "required": ["path"],
        },
    },
    {
        "name": "call_api",
        "description": "Call a validated API endpoint. Only call this after validate_api_endpoint returns valid.",
        "input_schema": {
            "type": "object",
            "properties": {
                "method": {"type": "string", "enum": ["GET", "POST", "PUT", "DELETE"]},
                "path": {"type": "string"},
                "body": {"type": "object"},
            },
            "required": ["method", "path"],
        },
    },
]

SYSTEM = """You are an API integration assistant.

<instructions>
- Before calling any API endpoint, ALWAYS call validate_api_endpoint first.
- If validation fails, do not proceed — explain the issue to the user.
- Never guess or invent endpoint paths.
</instructions>"""


def dispatch_tool(name: str, inputs: dict) -> str:
    if name == "validate_api_endpoint":
        valid, message = validate_endpoint(inputs["path"])
        return message
    elif name == "call_api":
        valid, _ = validate_endpoint(inputs["path"])
        if not valid:
            return f"Blocked: endpoint '{inputs['path']}' was not validated. Call validate_api_endpoint first."
        return f"API call simulated: {inputs['method']} {BASE_URL}{inputs['path']}"
    return "Unknown tool"


messages = [{"role": "user", "content": "Get the profile for user abc123"}]

while True:
    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=512,
        system=SYSTEM,
        tools=TOOLS,
        messages=messages,
    )

    messages.append({"role": "assistant", "content": response.content})

    if response.stop_reason == "end_turn":
        print(response.content[0].text)
        break

    tool_results = []
    for block in response.content:
        if block.type == "tool_use":
            result = dispatch_tool(block.name, block.input)
            print(f"[{block.name}] {block.input} → {result}")
            tool_results.append({"type": "tool_result", "tool_use_id": block.id, "content": result})

    messages.append({"role": "user", "content": tool_results})
```

**Expected Token Savings:** Validation gate catches hallucinated endpoints before they consume an HTTP round-trip and a model turn to process the 404.
**Environment:** Python 3.9+; regex allowlist scales to hundreds of endpoints.

### Option 3: OpenAPI spec loader — ground agent in machine-readable spec

```python
import json
import anthropic

client = anthropic.Anthropic()


def load_openapi_spec(spec: dict) -> dict[str, list[str]]:
    """
    Extract all valid path+method combinations from an OpenAPI 3.x spec.
    Returns: {"/v1/users": ["GET", "POST"], ...}
    """
    endpoints: dict[str, list[str]] = {}
    paths = spec.get("paths", {})
    for path, methods in paths.items():
        endpoints[path] = [m.upper() for m in methods if m.lower() in
                           ("get", "post", "put", "patch", "delete", "head", "options")]
    return endpoints


def format_endpoints_for_prompt(endpoints: dict[str, list[str]]) -> str:
    """Render endpoints as a compact reference for the system prompt."""
    lines = []
    for path, methods in sorted(endpoints.items()):
        lines.append(f"  {', '.join(methods)} {path}")
    return "\n".join(lines)


# Minimal OpenAPI spec (in production, load from file or URL)
OPENAPI_SPEC = {
    "openapi": "3.0.0",
    "info": {"title": "Example API", "version": "1.0.0"},
    "paths": {
        "/v1/users": {
            "get": {"summary": "List users"},
            "post": {"summary": "Create user"},
        },
        "/v1/users/{id}": {
            "get": {"summary": "Get user by ID"},
            "put": {"summary": "Update user"},
            "delete": {"summary": "Delete user"},
        },
        "/v1/users/{id}/orders": {
            "get": {"summary": "List orders for a user"},
        },
        "/v2/products": {
            "get": {"summary": "List products"},
        },
        "/v2/products/{sku}": {
            "get": {"summary": "Get product by SKU"},
        },
    },
}

valid_endpoints = load_openapi_spec(OPENAPI_SPEC)
endpoint_ref = format_endpoints_for_prompt(valid_endpoints)

SYSTEM = f"""You are an API assistant. Use only these endpoints:

<valid_endpoints>
{endpoint_ref}
</valid_endpoints>

Rules:
- Path parameters like {{id}} or {{sku}} are replaced with actual values.
- Never call endpoints not in the list above.
- If asked for an operation that has no matching endpoint, say so explicitly."""


def validate_against_spec(method: str, path: str) -> tuple[bool, str]:
    """Check method+path against loaded spec."""
    # Normalize path parameters
    import re
    normalized = re.sub(r"/[^/]+", lambda m: "/{param}" if not m.group().startswith("/{") else m.group(), path)

    # Try exact match first, then pattern match
    for spec_path, methods in valid_endpoints.items():
        spec_pattern = re.escape(spec_path).replace(r"\{[^}]+\}", "[^/]+")
        if re.fullmatch(spec_pattern.replace(r"\{", "{").replace(r"\}", "}").replace("{", "(?P<").replace("}", ">[^/]+)"), path.lstrip("/")):
            if method.upper() in methods:
                return True, f"Valid: {method} {path}"
            return False, f"Method {method} not allowed on {path}. Allowed: {', '.join(methods)}"

    # Try simple prefix-based match for common cases
    for spec_path, methods in valid_endpoints.items():
        spec_re = re.compile("^" + re.sub(r"\{[^}]+\}", r"[^/]+", spec_path) + "$")
        if spec_re.match(path):
            if method.upper() in methods:
                return True, f"Valid: {method} {path}"
            return False, f"Method {method} not allowed on {path}. Allowed: {', '.join(methods)}"

    return False, f"Endpoint '{path}' not found in API spec."


response = client.messages.create(
    model="claude-sonnet-4-6",
    max_tokens=512,
    system=SYSTEM,
    messages=[{"role": "user", "content": "How would I get the orders for user 42?"}],
)
print(response.content[0].text)

# Validate what the model suggests
ok, msg = validate_against_spec("GET", "/v1/users/42/orders")
print(f"\nValidation: {msg}")
```

**Expected Token Savings:** Machine-readable spec grounding eliminates hallucinated endpoints entirely.
**Environment:** Python 3.9+; works with any OpenAPI 3.x spec; load from YAML with `pyyaml`.

### Option 4: URL hallucination detector with Haiku classifier

```python
import re
import anthropic

client = anthropic.Anthropic()

KNOWN_APIS = {
    "stripe": {
        "base": "https://api.stripe.com",
        "prefixes": ["/v1/charges", "/v1/customers", "/v1/payment_intents", "/v1/subscriptions"],
    },
    "github": {
        "base": "https://api.github.com",
        "prefixes": ["/repos/", "/users/", "/orgs/", "/gists", "/search/"],
    },
    "openai": {
        "base": "https://api.openai.com",
        "prefixes": ["/v1/chat/completions", "/v1/embeddings", "/v1/models", "/v1/images/"],
    },
}


def extract_urls_from_text(text: str) -> list[str]:
    """Extract all URLs and endpoint paths from a text block."""
    # Match full URLs
    url_pattern = re.compile(r"https?://[^\s\"`']+")
    urls = url_pattern.findall(text)

    # Match bare API paths (starting with /v1, /v2, /api)
    path_pattern = re.compile(r"(?<!['\"`])/(?:v\d+|api)/[^\s\"`',)]+")
    paths = path_pattern.findall(text)

    return urls + paths


def classify_url_confidence(url: str, api_context: str) -> str:
    """Use Haiku to assess if a URL looks real or hallucinated."""
    response = client.messages.create(
        model="claude-haiku-4-5-20251001",
        max_tokens=60,
        system=(
            "You assess whether an API URL is likely real or hallucinated. "
            "Reply with exactly one word: REAL, HALLUCINATED, or UNCERTAIN."
        ),
        messages=[{
            "role": "user",
            "content": f"API context: {api_context}\nURL to assess: {url}",
        }],
    )
    return response.content[0].text.strip().upper()


def check_against_known_apis(url: str) -> tuple[str, str]:
    """Check URL against known API patterns."""
    for api_name, info in KNOWN_APIS.items():
        if info["base"] in url or any(prefix in url for prefix in info["prefixes"]):
            matched = [p for p in info["prefixes"] if p in url]
            if matched:
                return "KNOWN_VALID", f"Matches {api_name} pattern: {matched[0]}"
            return "KNOWN_BASE_ONLY", f"Base URL matches {api_name} but path is unrecognized"
    return "UNKNOWN", "URL does not match any known API"


def audit_response_for_url_hallucinations(response_text: str, api_context: str = "") -> list[dict]:
    urls = extract_urls_from_text(response_text)
    results = []

    for url in urls:
        known_status, known_msg = check_against_known_apis(url)
        if known_status == "KNOWN_VALID":
            results.append({"url": url, "status": "OK", "reason": known_msg})
            continue

        # Use Haiku to classify uncertain URLs
        haiku_verdict = classify_url_confidence(url, api_context)
        results.append({
            "url": url,
            "status": haiku_verdict,
            "reason": f"{known_msg}; Haiku verdict: {haiku_verdict}",
        })

    return results


# Test with a response containing a mix of real and hallucinated URLs
test_response = """
To charge a customer, use:
POST https://api.stripe.com/v1/charges

To search for payments:
GET https://api.stripe.com/v2/payment_search?q=foo

GitHub repo info is at:
GET https://api.github.com/repos/{owner}/{repo}
"""

print("URL audit results:")
for finding in audit_response_for_url_hallucinations(test_response, "Stripe payment API integration"):
    print(f"  [{finding['status']}] {finding['url']}")
    print(f"    → {finding['reason']}")
```

**Expected Token Savings:** Detecting hallucinated URLs before tool execution prevents failed round-trips; Haiku classification costs ~30 tokens vs. full retry cycles.
**Environment:** Python 3.9+; Haiku classifier adds ~50ms latency per URL; suitable for post-generation auditing.

### Option 5: Structured endpoint selection instead of free-form URL generation

```python
import anthropic

client = anthropic.Anthropic()

# Force the model to SELECT from a known list rather than GENERATE a URL
ENDPOINT_REGISTRY = [
    {"id": "list_users", "method": "GET", "path": "/v1/users", "description": "List all users with pagination"},
    {"id": "get_user", "method": "GET", "path": "/v1/users/{id}", "description": "Get a specific user by ID"},
    {"id": "create_user", "method": "POST", "path": "/v1/users", "description": "Create a new user"},
    {"id": "update_user", "method": "PUT", "path": "/v1/users/{id}", "description": "Update an existing user"},
    {"id": "delete_user", "method": "DELETE", "path": "/v1/users/{id}", "description": "Delete a user"},
    {"id": "list_orders", "method": "GET", "path": "/v1/orders", "description": "List orders with optional filters"},
    {"id": "get_order", "method": "GET", "path": "/v1/orders/{id}", "description": "Get a specific order"},
    {"id": "search", "method": "GET", "path": "/v2/search", "description": "Full-text search across all resources"},
]

ENDPOINT_IDS = [e["id"] for e in ENDPOINT_REGISTRY]
ENDPOINT_MAP = {e["id"]: e for e in ENDPOINT_REGISTRY}

# Tool schema forces selection from enum rather than free-form string
TOOLS = [
    {
        "name": "call_api",
        "description": "Call a known API endpoint. Select the endpoint_id from the allowed list.",
        "input_schema": {
            "type": "object",
            "properties": {
                "endpoint_id": {
                    "type": "string",
                    "enum": ENDPOINT_IDS,  # ← Critical: constrain to known values
                    "description": "The operation to perform",
                },
                "path_params": {
                    "type": "object",
                    "description": "Values for path parameters like {id}",
                },
                "query_params": {
                    "type": "object",
                    "description": "Query string parameters",
                },
                "body": {
                    "type": "object",
                    "description": "Request body for POST/PUT",
                },
            },
            "required": ["endpoint_id"],
        },
    }
]


def execute_api_call(endpoint_id: str, path_params: dict, query_params: dict, body: dict) -> str:
    endpoint = ENDPOINT_MAP[endpoint_id]
    path = endpoint["path"]

    # Substitute path parameters
    for key, value in (path_params or {}).items():
        path = path.replace(f"{{{key}}}", str(value))

    full_url = f"https://api.example.com{path}"
    return f"Simulated: {endpoint['method']} {full_url} (params: {query_params}, body: {body})"


messages = [{"role": "user", "content": "Get the details of order number 789"}]

while True:
    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=512,
        tools=TOOLS,
        messages=messages,
    )

    messages.append({"role": "assistant", "content": response.content})

    if response.stop_reason == "end_turn":
        print(next(b.text for b in response.content if b.type == "text"))
        break

    tool_results = []
    for block in response.content:
        if block.type == "tool_use":
            inp = block.input
            result = execute_api_call(
                inp["endpoint_id"],
                inp.get("path_params", {}),
                inp.get("query_params", {}),
                inp.get("body", {}),
            )
            print(f"API call: {result}")
            tool_results.append({"type": "tool_result", "tool_use_id": block.id, "content": result})

    messages.append({"role": "user", "content": tool_results})
```

**Expected Token Savings:** Enum-constrained tool input makes URL hallucination structurally impossible — zero retry cost from invalid endpoints.
**Environment:** Python 3.9+; enum constraint is the most reliable prevention strategy.

### Option 6: Retrieval-augmented endpoint lookup from docs

```python
import json
import anthropic

client = anthropic.Anthropic()

# Simulated endpoint documentation store (in production: vector DB)
ENDPOINT_DOCS = [
    {"endpoint": "GET /v1/customers", "description": "List all customers. Supports limit, offset, email filter.", "params": ["limit", "offset", "email"]},
    {"endpoint": "GET /v1/customers/{id}", "description": "Retrieve a single customer by their ID.", "params": ["id (path)"]},
    {"endpoint": "POST /v1/customers", "description": "Create a new customer. Body: name, email, phone.", "params": ["name", "email", "phone"]},
    {"endpoint": "GET /v1/invoices", "description": "List invoices. Filter by customer_id, status, date range.", "params": ["customer_id", "status", "from_date", "to_date"]},
    {"endpoint": "POST /v1/invoices/{id}/pay", "description": "Mark an invoice as paid.", "params": ["id (path)", "payment_method"]},
    {"endpoint": "GET /v1/products", "description": "List all products.", "params": ["category", "active"]},
    {"endpoint": "GET /v1/reports/revenue", "description": "Get revenue report by date range.", "params": ["from_date", "to_date", "group_by"]},
]


def find_relevant_endpoints(user_intent: str, top_k: int = 3) -> list[dict]:
    """
    Use Haiku to find the most relevant endpoints for a user's intent.
    In production, replace with embedding similarity search.
    """
    docs_json = json.dumps(ENDPOINT_DOCS, indent=2)
    response = client.messages.create(
        model="claude-haiku-4-5-20251001",
        max_tokens=256,
        system="You match user intents to API endpoints. Return a JSON array of the top matching endpoint strings only.",
        messages=[{
            "role": "user",
            "content": f"User intent: {user_intent}\n\nAvailable endpoints:\n{docs_json}\n\nReturn the top {top_k} endpoint strings as a JSON array.",
        }],
    )
    try:
        import re
        json_match = re.search(r"\[.*?\]", response.content[0].text, re.DOTALL)
        if json_match:
            endpoint_strings = json.loads(json_match.group())
            return [doc for doc in ENDPOINT_DOCS if doc["endpoint"] in endpoint_strings]
    except (json.JSONDecodeError, AttributeError):
        pass
    return ENDPOINT_DOCS[:top_k]  # Fallback: return first N


def build_system_with_retrieved_endpoints(user_intent: str) -> str:
    relevant = find_relevant_endpoints(user_intent)
    endpoint_ref = "\n".join(
        f"  {doc['endpoint']}\n    Params: {', '.join(doc['params'])}\n    {doc['description']}"
        for doc in relevant
    )
    return f"""You are an API assistant.

<relevant_endpoints>
{endpoint_ref}
</relevant_endpoints>

Use only endpoints listed above. State the exact endpoint before calling it."""


user_intent = "Show me all unpaid invoices for customer 42"
system = build_system_with_retrieved_endpoints(user_intent)
print("System prompt endpoint section:\n", system[system.find("<relevant"):system.find("</relevant>")+12])

response = client.messages.create(
    model="claude-sonnet-4-6",
    max_tokens=512,
    system=system,
    messages=[{"role": "user", "content": user_intent}],
)
print("\nResponse:", response.content[0].text)
```

**Expected Token Savings:** Retrieval injects only 2–3 relevant endpoints instead of full API spec, saving 200–2,000 tokens per turn while maintaining grounding.
**Environment:** Python 3.9+; replace keyword search with vector embeddings for semantic accuracy.
