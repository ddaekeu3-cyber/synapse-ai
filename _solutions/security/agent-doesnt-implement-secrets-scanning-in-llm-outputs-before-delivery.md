## Why This Happens

LLMs reproduce text from their context. If the agent injects retrieved documents, tool outputs, or system prompts that contain credentials, the model may reproduce those credentials verbatim — or paraphrase them in a way that still exposes sensitive material. Additionally, models trained on public code repositories have memorized patterns from leaked credentials and may generate plausible-looking but real secrets when asked for example configurations. Pre-delivery scanning intercepts the model's output before it reaches the user and applies pattern-based detection to identify and redact credential-like content.

## Solution 1: Secret Pattern Registry

import re
from dataclasses import dataclass, field
from enum import Enum
from typing import List, Optional, Pattern


class SecretSeverity(str, Enum):
    CRITICAL = "critical"   # real credential patterns (API keys, tokens)
    HIGH = "high"           # connection strings, private keys
    MEDIUM = "medium"       # potential credentials, ambiguous patterns
    LOW = "low"             # generic high-entropy strings


@dataclass
class SecretPattern:
    name: str
    pattern: str
    severity: SecretSeverity
    description: str
    redact_strategy: str = "full"    # "full" | "partial" | "alert_only"
    compiled: Optional[re.Pattern] = field(default=None, init=False)

    def __post_init__(self) -> None:
        self.compiled = re.compile(self.pattern, re.IGNORECASE | re.MULTILINE)


def default_secret_patterns() -> List[SecretPattern]:
    return [
        SecretPattern(
            name="openai_api_key",
            pattern=r"sk-[A-Za-z0-9]{20,}",
            severity=SecretSeverity.CRITICAL,
            description="OpenAI API key",
        ),
        SecretPattern(
            name="anthropic_api_key",
            pattern=r"sk-ant-[A-Za-z0-9\-_]{20,}",
            severity=SecretSeverity.CRITICAL,
            description="Anthropic API key",
        ),
        SecretPattern(
            name="github_pat",
            pattern=r"gh[ps]_[A-Za-z0-9]{36}",
            severity=SecretSeverity.CRITICAL,
            description="GitHub personal access token",
        ),
        SecretPattern(
            name="aws_access_key",
            pattern=r"AKIA[0-9A-Z]{16}",

## Solution 2: Secret Finding

```python
from dataclasses import dataclass
from typing import List


@dataclass
class SecretFinding:
    pattern_name: str
    severity: SecretSeverity
    match_start: int
    match_end: int
    matched_text: str
    redact_strategy: str
    description: str

    def preview(self, context_chars: int = 10) -> str:
        prefix = self.matched_text[:min(context_chars, len(self.matched_text))]
        return f"{prefix}..." if len(self.matched_text) > context_chars else self.matched_text
```

## Solution 3: Output Secrets Scanner

```python
from typing import List, Tuple


class OutputSecretsScanner:
    """
    Scans LLM output text for secret patterns.
    Returns all findings sorted by position in the text.
    """

    def __init__(self, patterns: List[SecretPattern] = None):
        self._patterns = patterns or default_secret_patterns()

    def scan(self, text: str) -> List[SecretFinding]:
        findings = []
        for pattern in self._patterns:
            for match in pattern.compiled.finditer(text):
                findings.append(SecretFinding(
                    pattern_name=pattern.name,
                    severity=pattern.severity,
                    match_start=match.start(),
                    match_end=match.end(),
                    matched_text=match.group(),
                    redact_strategy=pattern.redact_strategy,
                    description=pattern.description,
                ))
        findings.sort(key=lambda f: f.match_start)
        return findings

    def has_critical(self, text: str) -> bool:
        for finding in self.scan(text):
            if finding.severity == SecretSeverity.CRITICAL:
                return True
        return False
```

## Solution 4: Output Redactor

```python
import re
from typing import List, Tuple


class OutputRedactor:
    """
    Applies redaction to LLM output based on secret findings.
    Supports full redaction, partial masking, and alert-only modes.
    Processes findings from right to left to preserve position indices.
    """

    REDACTED = "[REDACTED]"

    def redact(self, text: str, findings: List[SecretFinding]) -> Tuple[str, int]:
        """Returns (redacted_text, redaction_count)."""
        if not findings:
            return text, 0

        redacted_count = 0
        result = list(text)

        # Process right to left to keep indices valid
        for finding in sorted(findings, key=lambda f: f.match_start, reverse=True):
            if finding.redact_strategy == "alert_only":
                continue   # don't modify text, just flag

            start, end = finding.match_start, finding.match_end

            if finding.redact_strategy == "full":
                replacement = self.REDACTED
            elif finding.redact_strategy == "partial":
                matched = finding.matched_text
                keep = min(6, len(matched) // 4)
                replacement = matched[:keep] + "***" + self.REDACTED
            else:
                replacement = self.REDACTED

            result[start:end] = list(replacement)
            redacted_count += 1

        return "".join(result), redacted_count
```

## Solution 5: Pre-Delivery Output Guard

```python
import time
from enum import Enum
from typing import Any, Callable, List, Optional


class DeliveryDecision(str, Enum):
    DELIVER = "deliver"
    DELIVER_REDACTED = "deliver_redacted"
    BLOCK = "block"


class PreDeliveryOutputGuard:
    """
    Scans LLM output before delivery, applies redaction, and
    decides whether to deliver the (possibly redacted) response
    or block it entirely.
    """

    def __init__(
        self,
        scanner: OutputSecretsScanner,
        redactor: OutputRedactor,
        block_on_critical: bool = True,
        audit_fn: Optional[Callable[[dict], None]] = None,
    ):
        self._scanner = scanner
        self._redactor = redactor
        self._block_critical = block_on_critical
        self._audit = audit_fn or (lambda ev: None)
        self._scanned = 0
        self._blocked = 0
        self._redacted = 0

    def guard(self, text: str, session_id: str = "") -> dict:
        self._scanned += 1
        findings = self._scanner.scan(text)

        has_critical = any(f.severity == SecretSeverity.CRITICAL for f in findings)
        has_actionable = any(f.redact_strategy != "alert_only" for f in findings)

        if has_critical and self._block_critical:
            self._blocked += 1
            self._audit({
                "event": "output_blocked",
                "session_id": session_id,
                "finding_count": len(findings),
                "critical_patterns": [f.pattern_name for f in findings if f.severity == SecretSeverity.CRITICAL],
                "timestamp": time.time(),
            })
            return {
                "decision": DeliveryDecision.BLOCK,
                "text": None,
                "findings": len(findings),
                "reason": "critical secret detected in output",
            }

        if has_actionable:
            redacted_text, count = self._redactor.redact(text, findings)
            self._redacted += 1
            self._audit({
                "event": "output_redacted",
                "session_id": session_id,
                "redaction_count": count,
                "patterns": [f.pattern_name for f in findings],
                "timestamp": time.time(),
            })
            return {
                "decision": DeliveryDecision.DELIVER_REDACTED,
                "text": redacted_text,
                "findings": len(findings),
                "redactions": count,
            }

        return {
            "decision": DeliveryDecision.DELIVER,
            "text": text,
            "findings": len(findings),
        }

    def stats(self) -> dict:
        return {
            "scanned": self._scanned,
            "blocked": self._blocked,
            "redacted": self._redacted,
        }
```

## Solution 6: Secrets Scan Audit Dashboard

```python
import time
from collections import Counter
from threading import Lock
from typing import List


class SecretsScanAuditDashboard:
    """
    Accumulates guard events and surfaces scan rate, pattern frequency,
    and block/redact ratios for security operations monitoring.
    """

    def __init__(self, guard: PreDeliveryOutputGuard, scanner: OutputSecretsScanner):
        self._guard = guard
        self._scanner = scanner
        self._events: List[dict] = []
        self._lock = Lock()

    def record_event(self, event: dict) -> None:
        with self._lock:
            self._events.append({**event, "recorded_at": time.time()})
            if len(self._events) > 10000:
                self._events.pop(0)

    def render(self, window_seconds: float = 3600.0) -> dict:
        cutoff = time.time() - window_seconds
        with self._lock:
            recent = [e for e in self._events if e.get("recorded_at", 0) >= cutoff]

        pattern_counts: Counter = Counter()
        for event in recent:
            for p in event.get("patterns", []) + event.get("critical_patterns", []):
                pattern_counts[p] += 1

        return {
            "generated_at": time.time(),
            "window_seconds": window_seconds,
            "guard_stats": self._guard.stats(),
            "recent_events": len(recent),
            "top_detected_patterns": pattern_counts.most_common(10),
        }
```

## Comparison

| Approach | Pattern Detection | Redaction | Block Decision | Audit Trail | Dashboard |
|---|---|---|---|---|---|
| OutputSecretsScanner | Yes (multi-pattern) | No | No | No | No |
| OutputRedactor | No | Yes (full/partial) | No | No | No |
| PreDeliveryOutputGuard | Via scanner | Via redactor | Yes (critical) | Yes | No |
| SecretsScanAuditDashboard | No | No | No | Yes | Yes |

**Best for production**: Set `block_on_critical=True` for all production deployments — a blocked response is always safer than delivering a credential. Use `redact_strategy="partial"` for medium-severity findings so users see that content was present but not the value itself, preventing confusion about missing context. Run `SecretsScanAuditDashboard` and alert when `blocked` count is non-zero in a 5-minute window — any block event indicates a prompt injection or retrieval pipeline that is exposing credentials to the model, which requires investigation regardless of whether the output was intercepted.
