## Why This Happens

Tool arguments are generated by the LLM based on user input and conversation context. If the user (or an attacker via prompt injection) manipulates the LLM into generating malicious argument values, those values reach tool implementations without further validation. A file reading tool that accepts `{"path": "../../../etc/shadow"}` will read the shadow file if the implementation does not validate the path. The fix requires validating arguments at the dispatch layer — between the LLM output and the tool implementation — using per-tool argument schemas that define what values are acceptable.

## Solution 1: Argument Injection Pattern Library

```python
import re
from dataclasses import dataclass
from enum import Enum
from typing import List, Pattern


class InjectionCategory(str, Enum):
    PATH_TRAVERSAL = "path_traversal"
    COMMAND_INJECTION = "command_injection"
    SQL_INJECTION = "sql_injection"
    SSRF = "ssrf"                         # Server-Side Request Forgery
    TEMPLATE_INJECTION = "template_injection"
    LDAP_INJECTION = "ldap_injection"


@dataclass
class InjectionPattern:
    category: InjectionCategory
    name: str
    pattern: str
    severity: int   # 1–10
    compiled: re.Pattern = None

    def __post_init__(self) -> None:
        self.compiled = re.compile(self.pattern, re.IGNORECASE | re.DOTALL)


def default_injection_patterns() -> List[InjectionPattern]:
    return [
        # Path traversal
        InjectionPattern(InjectionCategory.PATH_TRAVERSAL, "dotdot_slash",
                         r"\.\.[/\\]", severity=9),
        InjectionPattern(InjectionCategory.PATH_TRAVERSAL, "absolute_path_sensitive",
                         r"^/?(etc/|proc/|sys/|dev/|root/)", severity=9),
        InjectionPattern(InjectionCategory.PATH_TRAVERSAL, "encoded_traversal",
                         r"%2e%2e[%2f%5c]|\.\.%2f|\.\.%5c", severity=9),

        # Command injection
        InjectionPattern(InjectionCategory.COMMAND_INJECTION, "shell_metachar",
                         r"[;&|`$(){}<>]|\$\(|`[^`]+`", severity=8),
        InjectionPattern(InjectionCategory.COMMAND_INJECTION, "newline_injection",
                         r"[\r\n]", severity=7),

        # SQL injection
        InjectionPattern(InjectionCategory.SQL_INJECTION, "sql_keywords",
                         r"\b(UNION\s+SELECT|DROP\s+TABLE|INSERT\s+INTO|DELETE\s+FROM|;.*--)\b",
                         severity=9),
        InjectionPattern(InjectionCategory.SQL_INJECTION, "comment_sequences",
                         r"(--\s|/\*|\*/|#\s)", severity=6),

        # SSRF
        InjectionPattern(InjectionCategory.SSRF, "private_ip",
                         r"(^|\W)(127\.\d+\.\d+\.\d+|10\.\d+\.\d+\.\d+|192\.168\.\d+\.\d+|169\.254\.\d+\.\d+|::1|localhost)",
                         severity=8),
        InjectionPattern(InjectionCategory.SSRF, "file_schema",
                         r"^file://", severity=9),
        InjectionPattern(InjectionCategory.SSRF, "internal_metadata",
                         r"169\.254\.169\.254|metadata\.google\.internal", severity=10),

        # Template injection
        InjectionPattern(InjectionCategory.TEMPLATE_INJECTION, "jinja_syntax",
                         r"\{\{.+\}\}|\{%.+%\}", severity=7),
        InjectionPattern(InjectionCategory.TEMPLATE_INJECTION, "mako_syntax",

    ]
```

## Solution 2: Argument Injection Detector

```python
from dataclasses import dataclass
from typing import Any, Dict, List


@dataclass
class InjectionFinding:
    argument_name: str
    argument_value: str
    pattern_name: str
    category: InjectionCategory
    severity: int
    matched_span: str


class ToolArgumentInjectionDetector:
    """
    Scans tool call arguments for injection patterns.
    Recursively checks nested dicts and lists.
    """

    def __init__(self, patterns: List[InjectionPattern]):
        self._patterns = patterns

    def scan(
        self, args: Dict[str, Any], path: str = ""
    ) -> List[InjectionFinding]:
        findings = []
        self._scan_recursive(args, path, findings)
        return findings

    def _scan_recursive(
        self, obj: Any, path: str, findings: List[InjectionFinding]
    ) -> None:
        if isinstance(obj, dict):
            for key, value in obj.items():
                self._scan_recursive(value, f"{path}.{key}" if path else key, findings)
        elif isinstance(obj, list):
            for i, item in enumerate(obj):
                self._scan_recursive(item, f"{path}[{i}]", findings)
        elif isinstance(obj, str):
            for pattern in self._patterns:
                match = pattern.compiled.search(obj)
                if match:
                    findings.append(InjectionFinding(
                        argument_name=path,
                        argument_value=obj[:200],
                        pattern_name=pattern.name,
                        category=pattern.category,
                        severity=pattern.severity,
                        matched_span=match.group()[:50],
                    ))
```

## Solution 3: Per-Tool Argument Allowlist

```python
import re
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional


@dataclass
class ArgumentConstraint:
    arg_name: str
    allowed_pattern: Optional[str] = None   # regex the value must match
    blocked_pattern: Optional[str] = None   # regex that blocks if matched
    max_length: int = 2048
    allowed_values: Optional[List[str]] = None


class ToolArgumentAllowlistValidator:
    """
    Validates tool arguments against per-tool allowlist constraints.
    Complements injection detection with positive validation.
    """

    def __init__(self, constraints: Dict[str, List[ArgumentConstraint]]):
        self._constraints = constraints

    def validate(
        self, tool_name: str, args: Dict[str, Any]
    ) -> List[str]:
        """Returns list of violation messages; empty = valid."""
        violations = []
        tool_constraints = self._constraints.get(tool_name, [])

        for constraint in tool_constraints:
            value = args.get(constraint.arg_name)
            if value is None:
                continue
            str_val = str(value)

            if len(str_val) > constraint.max_length:
                violations.append(
                    f"Argument '{constraint.arg_name}' exceeds max length "
                    f"({len(str_val)} > {constraint.max_length})"
                )

            if constraint.allowed_values and str_val not in constraint.allowed_values:
                violations.append(
                    f"Argument '{constraint.arg_name}' value '{str_val[:50]}' "
                    f"not in allowed values"
                )

            if constraint.allowed_pattern and not re.fullmatch(
                constraint.allowed_pattern, str_val
            ):
                violations.append(
                    f"Argument '{constraint.arg_name}' does not match "
                    f"required pattern '{constraint.allowed_pattern}'"
                )

            if constraint.blocked_pattern and re.search(
                constraint.blocked_pattern, str_val, re.IGNORECASE
            ):
                violations.append(
                    f"Argument '{constraint.arg_name}' matched blocked pattern"
                )

        return violations
```

## Solution 4: Injection-Blocking Dispatcher

```python
import time
from typing import Any, Callable, Dict, List


class InjectionBlockedError(Exception):
    def __init__(self, tool_name: str, findings: List[InjectionFinding]):
        super().__init__(
            f"Tool '{tool_name}' call blocked: injection detected in arguments — "
            + ", ".join(f"{f.argument_name}:{f.pattern_name}" for f in findings)
        )
        self.tool_name = tool_name
        self.findings = findings


class InjectionBlockingDispatcher:
    """
    Intercepts tool calls, scans arguments for injection patterns,
    and blocks dispatch if any high-severity finding is detected.
    """

    def __init__(
        self,
        detector: ToolArgumentInjectionDetector,
        allowlist_validator: ToolArgumentAllowlistValidator,
        block_severity_threshold: int = 6,
    ):
        self._detector = detector
        self._validator = allowlist_validator
        self._threshold = block_severity_threshold
        self._blocked_calls: list = []

    async def dispatch(
        self,
        tool_name: str,
        args: Dict[str, Any],
        tool_fn: Callable,
    ) -> Any:
        # Injection detection
        findings = self._detector.scan(args)
        high_severity = [f for f in findings if f.severity >= self._threshold]

        # Allowlist validation
        violations = self._validator.validate(tool_name, args)

        if high_severity or violations:
            self._blocked_calls.append({
                "ts": time.time(),
                "tool_name": tool_name,
                "findings": [f.pattern_name for f in high_severity],
                "violations": violations,
            })
            if high_severity:
                raise InjectionBlockedError(tool_name, high_severity)
            raise ValueError(f"Argument validation failed for '{tool_name}': " + "; ".join(violations))

        return await tool_fn(**args)

    def blocked_call_summary(self, last_n: int = 50) -> list:
        return self._blocked_calls[-last_n:]
```

## Solution 5: Injection Audit Logger

```python
import time
from typing import List


class InjectionAuditLogger:
    """
    Records injection attempts for security incident investigation
    and pattern tuning.
    """

    def __init__(self, max_records: int = 10000):
        self._max = max_records
        self._records: List[dict] = []

    def log_blocked(
        self,
        tool_name: str,
        args_preview: str,
        findings: List[InjectionFinding],
        session_id: str = "",
    ) -> None:
        if len(self._records) >= self._max:
            self._records.pop(0)
        self._records.append({
            "ts": time.time(),
            "event": "injection_blocked",
            "session_id": session_id,
            "tool_name": tool_name,
            "args_preview": args_preview[:300],
            "findings": [
                {
                    "arg": f.argument_name,
                    "category": f.category.value,
                    "pattern": f.pattern_name,
                    "severity": f.severity,
                }
                for f in findings
            ],
        })

    def summary(self, window_seconds: float = 3600.0) -> dict:
        cutoff = time.time() - window_seconds
        recent = [r for r in self._records if r["ts"] >= cutoff]
        from collections import Counter
        category_counts = Counter(
            f["category"] for r in recent for f in r["findings"]
        )
        return {
            "window_seconds": window_seconds,
            "blocked_calls": len(recent),
            "unique_sessions": len({r["session_id"] for r in recent}),
            "by_category": dict(category_counts),
            "top_tools_targeted": Counter(r["tool_name"] for r in recent).most_common(5),
        }
```

## Solution 6: Injection Detection Coverage Auditor

```python
from typing import List


class InjectionDetectionCoverageAuditor:
    """
    Reports which tools have allowlist constraints configured
    and which rely only on pattern detection.
    """

    def __init__(
        self,
        allowlist_validator: ToolArgumentAllowlistValidator,
        known_tools: List[str],
    ):
        self._validator = allowlist_validator
        self._known = known_tools

    def audit(self) -> dict:
        covered = set(self._validator._constraints.keys())
        all_tools = set(self._known)
        uncovered = all_tools - covered

        return {
            "total_tools": len(all_tools),
            "with_allowlist_constraints": len(covered & all_tools),
            "pattern_detection_only": sorted(uncovered),
            "coverage_pct": round(len(covered & all_tools) / max(len(all_tools), 1) * 100, 1),
            "recommendation": (
                "All tools have allowlist constraints" if not uncovered
                else f"Add ArgumentConstraint for: {', '.join(sorted(uncovered))}"
            ),
        }
```

## Comparison

| Approach | Pattern Detection | Positive Allowlist | Recursive Scan | Blocking | Audit Trail |
|---|---|---|---|---|---|
| ToolArgumentInjectionDetector | Yes (regex library) | No | Yes | No | No |
| ToolArgumentAllowlistValidator | No | Yes (per-tool) | No | No | No |
| InjectionBlockingDispatcher | Via detector | Via validator | Via detector | Yes | No |
| InjectionAuditLogger | No | No | No | No | Yes |
| InjectionDetectionCoverageAuditor | No | Via validator | No | No | Yes (coverage) |

**Best for production**: Run `InjectionBlockingDispatcher` with `block_severity_threshold=6` — this blocks all path traversal, SQL injection, command injection, and SSRF patterns (all severity ≥ 7) while allowing low-severity warnings to pass. Add `ArgumentConstraint` for every file-path argument with `allowed_pattern=r"^[a-zA-Z0-9_\-./]+$"` and `blocked_pattern=r"\.\."` as a positive+negative constraint pair. Export `InjectionAuditLogger.summary()` as a security metric daily: a spike in `blocked_calls` from a single session indicates an active prompt injection attack and should trigger session termination.
