## Why This Happens

LLMs generate tool call arguments as free-form text constrained only by the schema type annotation and their training. Prompt injections in retrieved content, user messages, or tool results can override the agent's intended behavior and cause the model to generate arguments containing shell metacharacters, path traversal sequences, SQL injection fragments, or URL-encoded bypass sequences. The tool implementation receives these as ordinary strings and executes them, because argument validation is typically absent — the schema only validates that the argument is a string, not that the string is safe. Injection prevention requires per-argument pattern validation that runs after schema validation and before tool execution.

## Solution 1: Argument Validation Rule

```python
import re
from dataclasses import dataclass, field
from enum import Enum
from typing import List, Optional, Pattern


class InjectionCategory(str, Enum):
    SHELL = "shell"
    PATH_TRAVERSAL = "path_traversal"
    SQL = "sql"
    URL = "url"
    TEMPLATE = "template"
    LDAP = "ldap"


@dataclass
class ArgumentValidationRule:
    parameter_name: str
    allowed_pattern: Optional[str] = None        # regex: value must match this
    blocked_patterns: List[str] = field(default_factory=list)  # regex: reject if any match
    max_length: int = 4096
    injection_categories: List[InjectionCategory] = field(default_factory=list)
    description: str = ""

    def compile_allowed(self) -> Optional[re.Pattern]:
        return re.compile(self.allowed_pattern) if self.allowed_pattern else None

    def compile_blocked(self) -> List[re.Pattern]:
        return [re.compile(p) for p in self.blocked_patterns]
```

## Solution 2: Injection Pattern Library

```python
from typing import Dict, List


_SHELL_METACHAR_PATTERN = r'[;&|`$(){}\\<>!]'
_PATH_TRAVERSAL_PATTERN = r'\.\.[\\/]|[\\/]\.\.|\.\.'
_SQL_INJECTION_PATTERN = (
    r"(?i)(\b(select|insert|update|delete|drop|create|alter|exec|union|having|where)\b"
    r"|'[^']*'|--\s|/\*|\*/|;\s*\w)"
)
_TEMPLATE_INJECTION_PATTERN = r'\{\{|\}\}|\{%|%\}|\$\{|\}\$'
_LDAP_INJECTION_PATTERN = r'[)(\\*\x00]'
_URL_SCHEME_PATTERN = r'(?i)(javascript:|data:|vbscript:|file:|ftp://)'


INJECTION_PATTERN_LIBRARY: Dict[InjectionCategory, str] = {
    InjectionCategory.SHELL: _SHELL_METACHAR_PATTERN,
    InjectionCategory.PATH_TRAVERSAL: _PATH_TRAVERSAL_PATTERN,
    InjectionCategory.SQL: _SQL_INJECTION_PATTERN,
    InjectionCategory.TEMPLATE: _TEMPLATE_INJECTION_PATTERN,
    InjectionCategory.LDAP: _LDAP_INJECTION_PATTERN,
    InjectionCategory.URL: _URL_SCHEME_PATTERN,
}


def build_rules_for_tool(tool_name: str) -> List[ArgumentValidationRule]:
    """
    Returns a default rule set for common tool types based on tool name hints.
    Override with per-tool configuration in production.
    """
    if "file" in tool_name or "read" in tool_name or "write" in tool_name:
        return [ArgumentValidationRule(
            parameter_name="path",
            blocked_patterns=[_PATH_TRAVERSAL_PATTERN, _SHELL_METACHAR_PATTERN],
            injection_categories=[InjectionCategory.PATH_TRAVERSAL, InjectionCategory.SHELL],
            description="File path must not contain traversal or shell chars",
        )]
    if "sql" in tool_name or "query" in tool_name or "db" in tool_name:
        return [ArgumentValidationRule(
            parameter_name="query",
            blocked_patterns=[_SQL_INJECTION_PATTERN],
            injection_categories=[InjectionCategory.SQL],
            description="SQL query must not contain injection fragments",
        )]
    if "shell" in tool_name or "exec" in tool_name or "run" in tool_name:
        return [ArgumentValidationRule(
            parameter_name="command",
            blocked_patterns=[_SHELL_METACHAR_PATTERN],
            injection_categories=[InjectionCategory.SHELL],
            description="Shell command must not contain metacharacters",
        )]
    return []
```

## Solution 3: Argument Injection Validator

```python
import re
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional


@dataclass
class ValidationViolation:
    parameter_name: str
    value_preview: str
    rule_description: str
    injection_category: Optional[str]
    severity: str = "error"


class ArgumentInjectionValidator:
    """
    Validates tool call arguments against registered rules.
    Detects injection patterns in string arguments before execution.
    """

    def __init__(self, rules: List[ArgumentValidationRule]):
        self._rules = {r.parameter_name: r for r in rules}

    def validate(
        self, args: Dict[str, Any]
    ) -> tuple[bool, List[ValidationViolation]]:
        violations: List[ValidationViolation] = []

        for param_name, value in args.items():
            rule = self._rules.get(param_name)
            if rule is None:
                # No rule for this param — apply universal checks
                if isinstance(value, str):
                    violations.extend(
                        self._universal_checks(param_name, value)
                    )
                continue

            if not isinstance(value, str):
                continue

            if len(value) > rule.max_length:
                violations.append(ValidationViolation(
                    parameter_name=param_name,
                    value_preview=value[:50] + "…",
                    rule_description=f"Exceeds max length ({rule.max_length})",
                    injection_category=None,
                ))

            # Allowed pattern check
            allowed_re = rule.compile_allowed()
            if allowed_re and not allowed_re.fullmatch(value):
                violations.append(ValidationViolation(
                    parameter_name=param_name,
                    value_preview=value[:50] + "…",
                    rule_description=f"Does not match allowed pattern: {rule.allowed_pattern}",
                    injection_category=None,
                ))

            # Blocked pattern checks
            for blocked_re in rule.compile_blocked():
                if blocked_re.search(value):
                    cat = rule.injection_categories[0].value if rule.injection_categories else "unknown"
                    violations.append(ValidationViolation(
                        parameter_name=param_name,
                        value_preview=value[:50] + "…",
                        rule_description=rule.description,
                        injection_category=cat,
                    ))
                    break

        return len(violations) == 0, violations

    @staticmethod
    def _universal_checks(param_name: str, value: str) -> List[ValidationViolation]:
        """Minimal checks applied to all string args without explicit rules."""
        violations = []
        # Null byte injection
        if "\x00" in value:
            violations.append(ValidationViolation(
                parameter_name=param_name,
                value_preview=repr(value[:30]),
                rule_description="Null byte detected in argument",
                injection_category="null_byte",
            ))
        return violations
```

## Solution 4: Validation-Gated Tool Dispatcher

```python
import asyncio
from typing import Any, Callable, Dict, List, Optional


class ValidationGatedToolDispatcher:
    """
    Intercepts tool calls, runs argument injection validation, and
    blocks execution if violations are detected.
    """

    def __init__(self, validator_registry: Dict[str, ArgumentInjectionValidator]):
        self._validators = validator_registry
        self._blocked_calls: int = 0
        self._total_calls: int = 0

    async def dispatch(
        self,
        tool_name: str,
        args: Dict[str, Any],
        tool_fn: Callable,
        session_id: str = "",
    ) -> Any:
        self._total_calls += 1

        validator = self._validators.get(tool_name)
        if validator:
            is_valid, violations = validator.validate(args)
            if not is_valid:
                self._blocked_calls += 1
                raise InjectionDetectedError(
                    tool_name=tool_name,
                    violations=violations,
                    session_id=session_id,
                )

        return await tool_fn(**args)

    def stats(self) -> dict:
        return {
            "total_calls": self._total_calls,
            "blocked_calls": self._blocked_calls,
            "block_rate_pct": round(
                self._blocked_calls / max(self._total_calls, 1) * 100, 2
            ),
        }


class InjectionDetectedError(Exception):
    def __init__(
        self,
        tool_name: str,
        violations: List[ValidationViolation],
        session_id: str = "",
    ):
        super().__init__(
            f"Injection detected in tool '{tool_name}': "
            + "; ".join(v.rule_description for v in violations)
        )
        self.tool_name = tool_name
        self.violations = violations
        self.session_id = session_id
```

## Solution 5: Injection Attempt Audit Logger

```python
import time
from typing import List


class InjectionAttemptAuditLogger:
    """
    Records every detected injection attempt for security review
    and pattern analysis. Surfaces which injection categories and
    tools are most frequently targeted.
    """

    def __init__(self, max_records: int = 10000):
        self._max = max_records
        self._records: List[dict] = []

    def record(self, error: InjectionDetectedError) -> None:
        if len(self._records) >= self._max:
            self._records.pop(0)
        self._records.append({
            "ts": time.time(),
            "tool_name": error.tool_name,
            "session_id": error.session_id,
            "violation_count": len(error.violations),
            "categories": list({v.injection_category for v in error.violations if v.injection_category}),
        })

    def summary(self, window_seconds: float = 3600.0) -> dict:
        cutoff = time.time() - window_seconds
        recent = [r for r in self._records if r["ts"] >= cutoff]
        by_category: dict = {}
        by_tool: dict = {}
        for r in recent:
            for cat in r["categories"]:
                by_category[cat] = by_category.get(cat, 0) + 1
            by_tool[r["tool_name"]] = by_tool.get(r["tool_name"], 0) + 1
        return {
            "window_seconds": window_seconds,
            "injection_attempts": len(recent),
            "by_category": by_category,
            "by_tool": by_tool,
        }
```

## Solution 6: Argument Injection Prevention Dashboard

```python
import time


class ArgumentInjectionPreventionDashboard:
    """
    Combines dispatcher block statistics and audit log summaries
    into a single security operational view.
    """

    def __init__(
        self,
        dispatcher: ValidationGatedToolDispatcher,
        audit_logger: InjectionAttemptAuditLogger,
    ):
        self._dispatcher = dispatcher
        self._audit = audit_logger

    def render(self) -> dict:
        return {
            "generated_at": time.time(),
            "dispatcher_stats": self._dispatcher.stats(),
            "injection_attempts_1h": self._audit.summary(3600.0),
            "injection_attempts_24h": self._audit.summary(86400.0),
        }
```

## Comparison

| Approach | Pattern-Based Validation | Per-Tool Rules | Universal Checks | Blocking | Audit |
|---|---|---|---|---|---|
| ArgumentInjectionValidator | Yes (allow + block) | Via rule registry | Yes (null byte) | No | No |
| ValidationGatedToolDispatcher | Via validator | Via validator | Via validator | Yes | No |
| InjectionAttemptAuditLogger | No | No | No | No | Yes |
| ArgumentInjectionPreventionDashboard | No | No | No | No | Yes (aggregate) |

**Best for production**: Register explicit `ArgumentValidationRule` objects for every tool that accepts file paths, shell commands, SQL queries, or URL inputs — do not rely solely on the universal null-byte check for these high-risk parameters. Use `allowed_pattern` (whitelist) rather than `blocked_patterns` (blacklist) wherever possible: an allowed pattern of `^[\w\-./]{1,256}$` for a file path rejects everything not on the list, while a blocked pattern can be bypassed with encoding or Unicode normalization. Log every `InjectionDetectedError` with the session ID: a session that triggers multiple injection attempts in quick succession is likely being driven by a prompt injection in retrieved content, not user error, and warrants session termination and content audit.
